const stageNodes    = document.querySelectorAll('.stage-node');

const statusTitle   = document.getElementById('timeline-status-title');

const statusBody    = document.getElementById('timeline-status-body');

const statusIcon    = document.getElementById('timeline-status-icon');

let animationTimers = []; // track timers so we can clear them on a new search

// Progress track only spans the space *between* the first and last node

// centers, so we map stage index (0-3) to a 0–100% fill of that span.

function progressForStage(index) {

  return (index / (STAGES.length - 1)) * 100;

}

// Paint the circles/labels up to (and including) `activeIndex`.

// `activeIndex` is fully "complete" only when `settled` is true (used so

// the very last stage can show as "active" briefly before settling green).

function renderStages(activeIndex, settled) {

  stageNodes.forEach((node) => {

    const i = Number(node.dataset.index);

    const circle = node.querySelector('.stage-circle');

    const timeEl = node.querySelector('.stage-time');

    circle.classList.remove('stage-pending', 'stage-active', 'stage-complete');

    if (i < activeIndex || (i === activeIndex && settled)) {

      circle.classList.add('stage-complete');

      circle.innerHTML = `<svg class="w-4 h-4" fill="none" stroke="currentColor" stroke-width="3" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" d="M5 13l4 4L19 7"/></svg>`;

      timeEl.textContent = 'Done';

    } else if (i === activeIndex) {

      circle.classList.add('stage-active');

      circle.textContent = String(i + 1);

      timeEl.textContent = 'In progress';

    } else {

      circle.classList.add('stage-pending');

      circle.textContent = String(i + 1);

      timeEl.textContent = '\u00A0';

    }

  });

  timelineProgress.style.width = progressForStage(activeIndex) + '%';

  const stage = STAGES[activeIndex];

  statusTitle.textContent = stage.title;

  statusBody.textContent = stage.body;

  // Swap the status icon + accent color once the flow fully completes

  if (activeIndex === STAGES.length - 1 && settled) {

    statusIcon.className = 'w-8 h-8 rounded-full bg-teal/10 flex items-center justify-center shrink-0 mt-0.5';

    statusIcon.innerHTML = `<svg class="w-4 h-4 text-teal" fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" d="M5 13l4 4L19 7"/></svg>`;

  } else {

    statusIcon.className = 'w-8 h-8 rounded-full bg-sunset/10 flex items-center justify-center shrink-0 mt-0.5';

    statusIcon.innerHTML = `<svg class="w-4 h-4 text-sunset" fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" d="M13 10V3L4 14h7v7l9-11h-7z"/></svg>`;

  }

}

// Auto-play through every stage — the "visual simulation" for WAKA-REFUND

function playFullSimulation() {

  animationTimers.forEach(clearTimeout);

  animationTimers = [];

  renderStages(0, false); // start at stage 1, "in progress"

  const delays = [1300, 1900, 1900]; // pacing between each subsequent stage

  let elapsed = 0;

  for (let i = 1; i < STAGES.length; i++) {

    elapsed += delays[i - 1];

    animationTimers.push(setTimeout(() => {

      const isLast = i === STAGES.length - 1;

      renderStages(i, isLast);

    }, elapsed));

  }

}

// A generic reference: show a believable static snapshot mid-flow

function showStaticSnapshot() {

  animationTimers.forEach(clearTimeout);

  animationTimers = [];

  renderStages(1, false); // sits at "Airline approved", in progress

}

pnrForm.addEventListener('submit', function (e) {

  e.preventDefault();

  const value = pnrInput.value.trim();

  if (!value) {

    pnrError.classList.remove('hidden');

    pnrForm.classList.add('shake');

    setTimeout(() => pnrForm.classList.remove('shake'), 500);

    return;

  }

  pnrError.classList.add('hidden');

  // Reset and show a brief "looking up" state before results appear

  timelineWrap.classList.add('hidden');

  pnrLoading.classList.remove('hidden');

  pnrLoading.classList.add('flex');

  setTimeout(() => {

    pnrLoading.classList.add('hidden');

    pnrLoading.classList.remove('flex');

    timelineWrap.classList.remove('hidden');

    if (value.toUpperCase() === 'WAKA-REFUND') {

      playFullSimulation();

    } else {

      showStaticSnapshot();

    }

  }, 900);

});

const chatLauncher   = document.getElementById('chat-launcher');

const chatPanel      = document.getElementById('chat-panel');

const iconOpen        = document.getElementById('chat-icon-open');

const iconClose        = document.getElementById('chat-icon-close');

const chatLog         = document.getElementById('chat-log');

const chatForm        = document.getElementById('chat-form');

const chatInput       = document.getElementById('chat-input');

let chatStarted = false;

function toggleChat() {

  const isHidden = chatPanel.classList.contains('chat-hidden');

  chatPanel.classList.toggle('chat-hidden');

  iconOpen.classList.toggle('hidden');

  iconClose.classList.toggle('hidden');

  // Send a one-time greeting the first time the chat is opened

  if (isHidden && !chatStarted) {

    chatStarted = true;

    setTimeout(() => {

      addBotMessage("Hi Chidinma \u{1F44B} I'm here to help with bookings, refunds, or anything travel-related. Pick an option below or type your question.");

    }, 300);

  }

}

chatLauncher.addEventListener('click', toggleChat);

function scrollChatToBottom() {

  chatLog.scrollTop = chatLog.scrollHeight;

}

function addUserMessage(text) {

  const el = document.createElement('div');

  el.className = 'flex justify-end';

  el.innerHTML = `<div class="bg-navy text-white rounded-2xl rounded-br-sm px-3.5 py-2.5 max-w-[80%]">${text}</div>`;

  chatLog.appendChild(el);

  scrollChatToBottom();

}

function addBotMessage(text) {

  const el = document.createElement('div');

  el.className = 'flex justify-start';

  el.innerHTML = `<div class="bg-white border border-black/5 rounded-2xl rounded-bl-sm px-3.5 py-2.5 max-w-[80%] text-navy/80">${text}</div>`;

  chatLog.appendChild(el);

  scrollChatToBottom();

}

// Shows animated dots, then replaces them with the bot's real message

function addBotMessageWithDelay(text, delay = 800) {

  const typingEl = document.createElement('div');

  typingEl.className = 'flex justify-start';

  typingEl.innerHTML = `

    <div class="bg-white border border-black/5 rounded-2xl rounded-bl-sm px-4 py-3 flex gap-1 items-center">

      <span class="typing-dot w-1.5 h-1.5 rounded-full bg-navy/30 inline-block"></span>

      <span class="typing-dot w-1.5 h-1.5 rounded-full bg-navy/30 inline-block"></span>

      <span class="typing-dot w-1.5 h-1.5 rounded-full bg-navy/30 inline-block"></span>

    </div>`;

  chatLog.appendChild(typingEl);

  scrollChatToBottom();

  setTimeout(() => {

    typingEl.remove();

    addBotMessage(text);

  }, delay);

}
